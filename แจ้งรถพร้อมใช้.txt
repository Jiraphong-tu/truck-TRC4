        // Image Upload & Preview Handler
        function handleImagePreview(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    uploadedImageData = e.target.result;
                    document.getElementById('image-preview').src = uploadedImageData;
                    document.getElementById('image-preview-container').classList.remove('hidden');

        // GPS Geolocation Handler
        function getGPSLocation() {
            const statusEl = document.getElementById('gps-status');
            if ("geolocation" in navigator) {
                statusEl.innerText = "กำลังค้นหาพิกัด...";
                navigator.geolocation.getCurrentPosition(
                    function(position) {
                        const lat = position.coords.latitude.toFixed(4);
                        const lng = position.coords.longitude.toFixed(4);
                        currentGPS = `${lat}, ${lng}`;
                        statusEl.innerText = `พิกัด: ${currentGPS}`;
                        statusEl.className = "text-[10px] text-emerald-400 font-semibold";
                        showToast("สำเร็จ", "ดึงพิกัด GPS เรียบร้อยแล้ว", "success");
                    },
                    function(error) {
                        statusEl.innerText = "ไม่สามารถดึงพิกัดได้ (โปรดเปิด GPS)";
                        statusEl.className = "text-[10px] text-red-400";
                        showToast("ข้อผิดพลาด", "ไม่สามารถเข้าถึงตำแหน่ง GPS ได้", "error");
                    }
                );
            } else {
                statusEl.innerText = "อุปกรณ์ไม่รองรับ GPS";
            }
        }

        // Calculate Metrics and Render Dashboard
        function updateDashboardMetrics() {
            const total = fleetData.length;
            const ready = fleetData.filter(f => f.status === 'ready').length;
            const repair = fleetData.filter(f => f.status === 'repair').length;
            const pending = fleetData.filter(f => f.status === 'pending').length;
            const breakdown = fleetData.filter(f => f.status === 'breakdown').length;

            document.getElementById('cnt-ready').innerText = ready;
            document.getElementById('cnt-repair').innerText = repair;
            document.getElementById('cnt-pending').innerText = pending;
            document.getElementById('cnt-breakdown').innerText = breakdown;

            const availabilityRate = Math.round((ready / total) * 100) || 0;
            document.getElementById('fleet-availability-rate').innerText = `${availabilityRate}%`;

            // Update Progress Bar
            document.getElementById('bar-ready').style.width = `${(ready / total) * 100}%`;
            document.getElementById('bar-repair').style.width = `${((repair + pending) / total) * 100}%`;
            document.getElementById('bar-breakdown').style.width = `${(breakdown / total) * 100}%`;
        }

        // Helper Toast Notification
        function showToast(title, msg, type = 'info') {
            const toast = document.getElementById('toast');
            const icon = document.getElementById('toast-icon');
            const titleEl = document.getElementById('toast-title');
            const msgEl = document.getElementById('toast-msg');

            titleEl.innerText = title;
            msgEl.innerText = msg;

            if (type === 'success') {
                icon.innerHTML = '<i class="fa-solid fa-circle-check text-emerald-400"></i>';
            } else if (type === 'error') {
                icon.innerHTML = '<i class="fa-solid fa-circle-xmark text-red-400"></i>';

            setTimeout(() => {
                toast.classList.add('hidden');
            }, 3000);
        }
